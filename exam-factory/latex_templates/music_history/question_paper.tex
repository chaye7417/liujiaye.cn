% 标准选项（不记录答案）
\newcommand{\xuanxiang}[3]{\begin{choices}
		\item #1
		\item #2
		\item #3
\end{choices}}

% 固定顺序选项（记录答案为 A）
\newcommand{\xuanxiangguding}[3]{%
  \begin{choices}
    \item #1
    \item #2
    \item #3
  \end{choices}
  \addanswer{A}%
}

% 随机选项（自动打乱顺序并记录正确答案）
\newcommand{\xuanxiangsuiji}[3]{%
  \pgfmathsetmacro{\randcase}{int(1+3*rnd)}%
  \ifcase\randcase
  \or % 情况1: A-B-C (原始顺序)
    \begin{choices}
      \item #1
      \item #2
      \item #3
    \end{choices}
    \addanswer{A}%
  \or % 情况2: B-C-A
    \begin{choices}
      \item #2
      \item #3
      \item #1
    \end{choices}
    \addanswer{C}%
  \or % 情况3: C-A-B
    \begin{choices}
      \item #3
      \item #1
      \item #2
    \end{choices}
    \addanswer{B}%
  \fi
}

% 读取题目的命令
\newcommand{\readquestions}[2]{
    \begin{QsNum}
    \DTLforeach*[\DTLisgt{\tid}{#1}\and\DTLislt{\tid}{#2}]{questions}{%
        \tid=题号,%
        \tiwenti=问题,%
        \optionA=选项A,%
        \optionB=选项B,%
        \optionC=选项C,%
        \answer=正确答案%
    }{%
        \item \nandu{\mydifficulty}\tiyuan{\mytopic}
        \tigan{\tiwenti}
        \ifthenelse{\boolean{shuffleoptions}}{%
          \xuanxiangsuiji{\optionA}{\optionB}{\optionC}%
        }{%
          \xuanxiangguding{\optionA}{\optionB}{\optionC}%
        }
    }
    \end{QsNum}
}
