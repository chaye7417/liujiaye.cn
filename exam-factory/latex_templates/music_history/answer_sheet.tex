% ==================== 水印设置 ====================
\newcommand{\mywatermark}{内部资料}
\newcommand{\mywatermarkcolor}{gray}
\newcommand{\mywatermarkopacity}{0.1}
\newboolean{showwatermark}
\setboolean{showwatermark}{true}

\newcommand{\AddWatermark}{%
  \ifthenelse{\boolean{showwatermark}}{%
    \begin{tikzpicture}[remember picture, overlay]
      \foreach \i in {0,1,...,10} {
        \foreach \j in {0,1,...,15} {
          \node [rotate=45, text=\mywatermarkcolor, opacity=\mywatermarkopacity, font=\bfseries\Large] 
          at ([xshift=\i*3cm, yshift=\j*2cm]current page.south west) {\mywatermark};
        }
      }
    \end{tikzpicture}
  }{}%
}

\AtBeginShipout{\AddWatermark}

\newcommand{\updateWatermarkSettings}[3]{
  \renewcommand{\mywatermark}{#1}
  \renewcommand{\mywatermarkcolor}{#2}
  \renewcommand{\mywatermarkopacity}{#3}
  
  \ifthenelse{\equal{#1}{}}{
    \setboolean{showwatermark}{false}
  }{
    \setboolean{showwatermark}{true}
  }
}

      \node[anchor=north,inner sep=0pt,outer sep=0pt] at (current page.north) {%

% ==================== 装饰标题命令 ====================
\newcommand{\decoratedtitle}[1]{%
  \thispagestyle{plain}%
  \pagebg
  \begin{center}
    {\fontsize{22}{22}\selectfont\bfseries
     \begin{tikzpicture}[baseline]
       % 白色描边效果
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0,0.07) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0.07,0) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0,-0.07) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (-0.07,0) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0.05,0.05) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (-0.05,0.05) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0.05,-0.05) {#1};
       \node[text=white, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (-0.05,-0.05) {#1};
       % 主文本
       \node[text=mycolor, align=center, font=\fontsize{22}{22}\selectfont\bfseries] at (0,0) {#1};
     \end{tikzpicture}
     \par}
  \end{center}
  \vspace{1em}
}
